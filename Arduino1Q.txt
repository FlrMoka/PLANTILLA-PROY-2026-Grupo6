content = """#include <Arduino_RouterBridge.h>
#include <Wire.h>
#include <U8g2lib.h>
U8G2_SSD1306_128X64_NONAME_F_HW_I2C u8g2(U8G2_R0, U8X8_PIN_NONE);
#define BUZZER_PIN 9
float temperatura = 0;
float rpm = 0;
bool alarma = false;
String estadoTemp = "SIN DATOS";
String estadoResp = "SIN DATOS";
const unsigned long TIMEOUT = 5000;
unsigned long ultimoPaquete = 0;
int rpmMinNormal = 30;
int rpmMaxNormal = 60;
void evaluarEstados() {
  alarma = false;
  if (temperatura<=33.9) { estadoTemp="HIPOT.SEV"; alarma=true; }
  else if (temperatura<=35.5) { estadoTemp="HIPOTERMIA"; alarma=true; }
  else if (temperatura<=37.9) { estadoTemp="NORMAL"; }
  else if (temperatura<=40.0) { estadoTemp="FIEBRE"; alarma=true; }
  else { estadoTemp="HIPERTERM"; alarma=true; }
  if (rpm<rpmMinNormal) { estadoResp="APNEA"; alarma=true; }
  else if (rpm>rpmMaxNormal) { estadoResp="TAQUIPNEA"; alarma=true; }
  else { estadoResp="NORMAL"; }
}
void actualizarPantalla() {
  bool sinDatos=(millis()-ultimoPaquete>TIMEOUT);
  u8g2.clearBuffer();
  u8g2.setFont(u8g2_font_6x10_tr);
  char buf[32];
  snprintf(buf, sizeof(buf), "Temp: %.1f C", sinDatos?0.0f:temperatura);
  u8g2.drawStr(0,10, sinDatos?"Temp: --.- C":buf);
  snprintf(buf, sizeof(buf), "RPM : %.0f", sinDatos?0.0f:rpm);
  u8g2.drawStr(0,20, sinDatos?"RPM : --":buf);
  u8g2.drawHLine(0,23,127);
  snprintf(buf, sizeof(buf), "Temp: %s", sinDatos?"SIN DATOS":estadoTemp.c_str());
  u8g2.drawStr(0,34, buf);
  snprintf(buf, sizeof(buf), "Resp: %s", sinDatos?"SIN DATOS":estadoResp.c_str());
  u8g2.drawStr(0,44, buf);
  u8g2.drawHLine(0,47,127);
  if (sinDatos) { u8g2.drawStr(0,57,"SIN CONEXION"); }
  else if (alarma) { u8g2.drawStr(0,57,"*** ALERTA ***"); }
  else { u8g2.drawStr(0,57,"MONITOR BEBE"); }
  u8g2.sendBuffer();
}
void actualizarDatos(float temp, float r) {
  temperatura=temp; rpm=r; ultimoPaquete=millis(); evaluarEstados();
}
void setup() {
  Serial.begin(115200);
  pinMode(BUZZER_PIN, OUTPUT);
  digitalWrite(BUZZER_PIN, LOW);
  Bridge.begin();
  Bridge.provide("actualizar_datos", actualizarDatos);
  u8g2.begin();
  u8g2.clearBuffer();
  u8g2.setFont(u8g2_font_6x10_tr);
  u8g2.drawStr(0,32,"Esperando datos...");
  u8g2.sendBuffer();
  ultimoPaquete=millis();
}
void loop() {
  if (alarma) { digitalWrite(BUZZER_PIN,(millis()/500)%2); }
  else { digitalWrite(BUZZER_PIN,LOW); }
  actualizarPantalla();
  delay(100);
}"""
with open('/home/arduino/ArduinoApps/proyecto/sketch/sketch.ino', 'w') as f:
    f.write(content)
