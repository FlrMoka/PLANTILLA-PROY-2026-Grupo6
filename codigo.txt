#include <Wire.h>
#include <Adafruit_Sensor.h>
#include <Adafruit_ADXL345_U.h>

#define ADXL345_CLK  12
#define ADXL345_MISO 10
#define ADXL345_MOSI 11
#define ADXL345_CS   8
const int pinSensor = A0;

Adafruit_ADXL345_Unified accel = Adafruit_ADXL345_Unified(ADXL345_CLK, ADXL345_MISO, ADXL345_MOSI, ADXL345_CS, 12345);

float lastY = 0;
unsigned long lastRespTime = 0;
float rpmPromedio = 0;
float sumaRPM = 0;
int conteoLecturas = 0;
unsigned long ventanaTiempo = 0;
unsigned long inicioTaquipnea = 0;
unsigned long inicioApnea = 0;
const unsigned long TIEMPO_PERSISTENCIA = 3000; 
float lastX = 0, lastZ = 0;
unsigned long lastCheckTime = 0;
bool movimientoValido = true;

void setup() {
  Serial.begin(9600);
  if(!accel.begin()) {
    while(1);
  }
  accel.setRange(ADXL345_RANGE_2_G);
  lastRespTime = millis();
  ventanaTiempo = millis();
}

void loop() {
  sensors_event_t event;
  accel.getEvent(&event);
  unsigned long currentTime = millis();

  if (currentTime - lastCheckTime >= 500) {
    float diffX = abs(event.acceleration.x - lastX);
    float diffY = abs(event.acceleration.y - lastY);
    float diffZ = abs(event.acceleration.z - lastZ);
    movimientoValido = (diffX < 1.50 && diffY < 1.50 && diffZ < 1.50);
    lastX = event.acceleration.x;
    lastZ = event.acceleration.z;
    lastCheckTime = currentTime;
  }

  float currentY = event.acceleration.y;
  float diffY_resp = abs(currentY - lastY);

  if (movimientoValido && diffY_resp >= 0.2) {
    if (currentTime - lastRespTime > 700) { 
      unsigned long timeGap = currentTime - lastRespTime;
      float rpmInstantanea = 60000.0 / timeGap;
      sumaRPM += rpmInstantanea;
      conteoLecturas++;
      lastRespTime = currentTime;
    }
  }

  if (currentTime - ventanaTiempo >= 1000) {
    int lecturaRaw = analogRead(pinSensor);
    float voltaje = lecturaRaw * (5000.0 / 1023.0);
    float temperaturaC = (voltaje - 1035.0) / -5.5;

    if (conteoLecturas > 0) {
      rpmPromedio = sumaRPM / conteoLecturas;
    } else if (currentTime - lastRespTime >= 15000) {
      rpmPromedio = 0; 
    }

    Serial.print("T: "); Serial.print(temperaturaC, 1); Serial.print("C (");
    if (temperaturaC <= 35.5) Serial.print("HIPOTERMIA");
    else if (temperaturaC >= 38.0) Serial.print("FIEBRE");
    else Serial.print("Normal");
    Serial.print(") | ");

    if (movimientoValido) {
      if (rpmPromedio > 60) {
        if (inicioTaquipnea == 0) inicioTaquipnea = currentTime;
        if (currentTime - inicioTaquipnea >= TIEMPO_PERSISTENCIA) {
          Serial.print("!!! TAQUIPNEA !!! | RPM: "); Serial.println(rpmPromedio);
        } else { Serial.print("Evaluando... | RPM: "); Serial.println(rpmPromedio); }
        inicioApnea = 0;
      } 
      else if (rpmPromedio < 30 || (currentTime - lastRespTime >= 15000)) {
        if (inicioApnea == 0) inicioApnea = currentTime;
        if (currentTime - inicioApnea >= TIEMPO_PERSISTENCIA) {
          Serial.print("!!! APNEA !!! | RPM: "); Serial.println(rpmPromedio);
        } else { Serial.print("Evaluando... | RPM: "); Serial.println(rpmPromedio); }
        inicioTaquipnea = 0;
      } 
      else {
        Serial.print("Resp: NORMAL | RPM: "); Serial.println(rpmPromedio);
        inicioTaquipnea = 0;
        inicioApnea = 0;
      }
    } else {
      Serial.println("MOVIMIENTO DETECTADO");
      inicioTaquipnea = 0;
      inicioApnea = 0;
    }

    sumaRPM = 0;
    conteoLecturas = 0;
    ventanaTiempo = currentTime;
  }

  lastY = currentY;
  delay(50); 
}
